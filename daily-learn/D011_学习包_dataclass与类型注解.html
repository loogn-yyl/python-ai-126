# 等价的手写版本（D008–D010 的你）
class Book:
    def __init__(self, title, price, count=1):
        self.title = title
        self.price = price
        self.count = count

    def __repr__(self):
        return f"Book(title={self.title!r}, price={self.price!r}, count={self.count!r})"

    def __eq__(self, other):
        if not isinstance(other, Book):
            return NotImplemented
        return (self.title, self.price, self.count) == (other.title, other.price, other.count)
注意 @dataclass 只生成「数据类」该有的东西。你自己写的方法（像 D008 的 line()）照样写在里面，不会被覆盖。它管的是「存数据 + 打印 + 比较」这三件重复劳动，业务逻辑还是你的。
字段顺序 = 参数顺序，而且「有默认值的必须排在没默认值的后面」。count: int = 1 写在 title: str 前面会直接 TypeError: non-default argument 'title' follows default argument——和普通函数的默认参数规则一模一样（D005 学过）。

2field()：可变默认值的正规解

D005 和 D008 都踩过这个坑：def f(x=[]) 只创建一次，所有调用共用。在 dataclass 里，Python 干脆不让你写：

from dataclasses import dataclass


@dataclass
class Bad:
    tags: list = []              # 想给个空列表当默认值
ValueError: mutable default <class 'list'> for field tags is not allowed: use default_factory

报错发生在「定义类的那一刻」，不是造对象的时候——连 import 都会失败。第一次遇到会懵，因为屏幕上根本没有你的运行代码。解决办法就写在报错里：use default_factory。

from dataclasses import dataclass, field


@dataclass
class Good:
    tags: list = field(default_factory=list)


a = Good()
a.tags.append("同学")
b = Good()
print(a.tags, b.tags, a.tags is b.tags)
['同学'] [] False
写法含义
tags: list = field(default_factory=list)每次造对象时调用一次 list()，人人拿到新的空列表
tags: list = []被明令禁止：一个列表被所有实例共享，就是 D008 那个 bug
field(default=0, repr=False)其他常用参数：default 普通默认值、repr=False 打印时不显示
为什么 dataclass 能拦、普通类拦不住？因为普通类的默认值是在 def __init__(self, tags=[]) 里求值的，Python 看不出来它可不可变；而 dataclass 的默认值是声明式的，它在定义类时就能检查类型。能在这个位置报错，就已经是它最大的价值之一。

3Enum：一组固定的值

通讯录的「状态」如果只用字符串，就永远不知道有几种合法取值，也挡不住 "actve" 这种拼错。用枚举定死：

from enum import Enum


class Status(Enum):
    ACTIVE = "active"
    BLOCKED = "blocked"


print(Status.ACTIVE)
print(Status.ACTIVE.name, Status.ACTIVE.value)
print(Status("active"))
print(Status.ACTIVE == "active")
print([s.name for s in Status])
Status.ACTIVE
ACTIVE active
Status.ACTIVE

['ACTIVE', 'BLOCKED']
写法结果说明
Status.ACTIVEStatus.ACTIVE打印出来带类名，一眼知道是哪一组的
.name / .value'ACTIVE' / 'active'名字给程序用，值给存储/接口用
Status("active")Status.ACTIVE从值反查成员；值不存在会抛 ValueError
== "active"False枚举成员不等于它的值，要比就比 .value
什么时候该用枚举：取值是有限几个、你在代码里要写 if 判断的时候——订单状态、星期、日志级别、文件类型。
什么时候不用：取值无限或者来自外部数据（姓名、城市名），那些用字符串就对了。
好处很实际：写了 if s == Status.PAID 之后，拼错成 Status.PADI 会立刻 AttributeError；而用字符串拼错 "padi" 只会让这个分支永远不成立，程序照跑，结果是错的——又是那个「不报错的 bug」。

4类型注解：写给人和 IDE 看的说明书

你可能已经在 D008 见过 def __init__(self, name: str, phone: str) 里的冒号了。那不是类型声明，是注解——Python 会把它存起来，但运行时不检查。

from typing import Optional


def find(name: str, data: dict[str, str]) -> Optional[str]:
    return data.get(name)


data = {"张三": "13800001111"}
print(find("张三", data))
print(find("李四", data))
print(find.__annotations__)
print("传 int 也不报错：", find(123, data))
13800001111
None
{'name': <class 'str'>, 'data': dict[str, str], 'return': typing.Optional[str]}
传 int 也不报错： None

最后一行是重点：传了 123 进去，没有任何报错，只是查不到东西返回了 None。注解只是被塞进 __annotations__ 这个字典里，没有任何强制力——这是 Python 的取舍：灵活优先，检查交给工具。

写法意思什么时候用
name: str应该是字符串所有函数参数和返回值
list[Contact]里面装的是 Contact 对象容器参数，比只写 list 有用得多
dict[str, int]键是字符串、值是整数同上
Optional[X]可能是 X，也可能是 None会返回 None 的函数必须标这个
X | None同上，3.10 之后的简写新代码可以这样写
注解真正的用处有三个，都不是「运行时检查」：
① IDE 补全：写完 c. VS Code 能列出 Contact 的字段，不用翻代码；
② Pylance 查错：传错类型会画波浪线（你装的扩展就是干这个的）；
③ 给人看：一个月后回来看 def find(contacts: list[Contact], name: str) -> Optional[Contact]，不用读函数体就知道怎么用、会不会返回 None。
别把注解当校验用。「用户输入必须是数字」这种事，仍然要用 D006 的 try/except 或者显式 if 判断。注解管不了外部数据——那是下一节 Pydantic 要解决的。

同一个思路（用类型描述数据），换一个会真的执行检查的库。这是业界用得最多的数据校验库，也是后面 RAG / Agent 阶段处理 API 返回、请求参数的标准工具。

6把通讯录的 Contact 换成 dataclass

D008 的 Contact 有 __init__、要手写 to_dict，还要记住哪些字段要存。换成 dataclass：

from dataclasses import dataclass, field, asdict


@dataclass
class Contact:
    name: str
    phone: str
    email: str = ""
    tags: list = field(default_factory=list)

    def line(self):
        parts = [self.name, self.phone]
        if self.email:
            parts.append(self.email)
        return "  ".join(parts)

    def to_dict(self):
        d = asdict(self)          # 一键转成字典
        d.pop("name")             # 但 name 在 JSON 里是 key，不重复存
        return d


a = Contact("张三", "13800001111", "zs@example.com")
b = Contact("张三", "13800001111", "zs@example.com")
print(a)
print("内容相等:", a == b)
print(a.line())
print(a.to_dict())

c = Contact("李四", "13900002222", tags=["同事"])
c.tags.append("常联系")
d = Contact("王五", "13700003333")

原来（D008）现在（D011）
手写 __init__ 三行五行字段声明，还是这么长，但多了类型信息
手写 __repr__ 才能打印自动有了，而且格式统一
自己实现 __eq__ 才相等自动有了，逐字段比
手写 to_dict() 逐个字段抄asdict(self) 一键，再 pop 掉不需要的
加一个字段要改两处加一行字段声明就行，asdict 自动带上
asdict 会把 name 也带上，所以要 pop 掉。这是设计问题不是 bug：JSON 里名字已经是 key 了，值里再存一遍就重复。什么时候该让对象自己决定存哪些字段——这正是 D008 说的「文件和内存是两件事」，今天只是把转换代码从 8 行缩到 3 行。
注意最后一行 ['同事', '常联系'] []——两个实例的 tags 互不影响，说明 default_factory 生效了。如果这里输出两个一模一样的列表，就是又踩了可变默认值的坑。

from dataclasses import dataclass
from typing import Optional


@dataclass
class Contact:
    name: str
    phone: str


def group_by_prefix(contacts: list[Contact]) -> dict[str, list[str]]:
    res: dict[str, list[str]] = {}
    for c in contacts:
        res.setdefault(c.phone[:3], []).append(c.name)
    return res


def find(contacts: list[Contact], name: str) -> Optional[Contact]:
    for c in contacts:
        if c.name == name:
            return c
    return None


data = [Contact("张三", "13800001111"), Contact("李四", "13900002222"), Contact("王五", "13800003333")]

import json
import os
from dataclasses import dataclass, field, asdict

FILE = "books.json"


@dataclass
class Book:
    title: str
    price: float
    tags: list = field(default_factory=list)

    def to_dict(self):
        return asdict(self)

    @classmethod
    def from_dict(cls, d):
        return cls(**d)


books = [Book("流畅的 Python", 128.0, ["进阶"]), Book("算法导论", 99.0)]

with open(FILE, "w", encoding="utf-8") as f:
    json.dump([b.to_dict() for b in books], f, ensure_ascii=False, indent=2)

with open(FILE, "r", encoding="utf-8") as f:
    back = [Book.from_dict(d) for d in json.load(f)]

print("读回来", len(back), "本")
print(back[0])
print("整体相等:", back == books)
print("tags 也相等:", back[0].tags == books[0].tags)
print("tags 不是同一个列表:", back[0].tags is not books[0].tags)
os.remove(FILE)

8今日笔记要点

# D011 笔记

## @dataclass
- 只写字段声明：name: str，__init__ / __repr__ / __eq__ 全自动
- 有默认值的字段必须排在没默认值的后面
- 自己写的方法照常写，不会被覆盖
- 加字段只改一行声明

## field()
- 可变默认值（list/dict/set）必须用 field(default_factory=list)
- 写成 tags: list = [] 会在【定义类的时候】就 ValueError，连 import 都失败
- 其他参数：default=、repr=False

## 常用开关
- @dataclass(order=True)   自动生成 __lt__ 等，可以 sorted()、可以 >
- @dataclass(frozen=True)  不可变；改属性抛 FrozenInstanceError；能 hash、能进 set

## Enum
- class Status(Enum): ACTIVE = "active"
- Status.ACTIVE 打印带类名；.name 是 'ACTIVE'，.value 是 'active'
- Status("active") 按值反查，值不存在抛 ValueError
- Status.ACTIVE == "active" 是 False ！要对比就 .value
- 适合：订单状态、星期、日志级别；不适合：姓名、城市名这类无限取值

## 类型注解
- def f(x: str) -> Optional[int]:   只是注解，运行时不检查！
- list[Contact] / dict[str, int]    容器要写里面的类型
- Optional[X] 或 X | None           可能返回 None 时必须标
- 用处：IDE 补全、Pylance 查错、一个月后自己看得懂
- 校验外部数据要靠 try/except 或 Pydantic，不能靠注解

## Pydantic（预览）
- 第三方库：uv run --with pydantic demo.py
- class Contact(BaseModel): 字段注解 + 自动校验 + 自动类型转换
- 脏数据抛 ValidationError，能说清是哪个字段、错在哪
- 一句话：dataclass 信任数据，Pydantic 不信任
